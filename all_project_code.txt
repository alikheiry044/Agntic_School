from sqlalchemy.orm import Session
from .models import Grade, Attendance


def check_grade_drop(db: Session, student_id: int):
    """
    نمرات هر درس رو بر اساس تاریخ مرتب می‌کنه و آخرین نمره رو
    با نمره قبلی مقایسه می‌کنه. اگه افت داشته باشه، گزارشش می‌کنه.
    """
    grades = (
        db.query(Grade)
        .filter(Grade.student_id == student_id)
        .order_by(Grade.subject, Grade.date)
        .all()
    )

    subjects = {}
    for g in grades:
        subjects.setdefault(g.subject, []).append(g)

    report = []
    for subject, records in subjects.items():
        if len(records) < 2:
            continue
        last = records[-1]
        previous = records[-2]
        diff = last.score - previous.score
        if diff < 0:
            report.append({
                "subject": subject,
                "previous_score": previous.score,
                "last_score": last.score,
                "change": diff,
                "status": "افت"
            })
        else:
            report.append({
                "subject": subject,
                "previous_score": previous.score,
                "last_score": last.score,
                "change": diff,
                "status": "پیشرفت" if diff > 0 else "ثابت"
            })

    return report


def check_unusual_absence(db: Session, student_id: int, threshold: int = 2):
    """
    اگه تعداد غیبت‌های دانش‌آموز از threshold بیشتر باشه،
    به‌عنوان غیبت غیرعادی گزارش می‌شه.
    """
    absences = (
        db.query(Attendance)
        .filter(Attendance.student_id == student_id, Attendance.status == "absent")
        .all()
    )

    return {
        "total_absences": len(absences),
        "is_unusual": len(absences) >= threshold,
        "dates": [a.date for a in absences],
    }

from .models import Student


def get_school_wide_report(db: Session):
    """
    روی همه دانش‌آموزان مدرسه اجرا می‌شه و دانش‌آموزانی که
    افت تحصیلی یا غیبت غیرعادی دارن رو مشخص می‌کنه.
    """
    students = db.query(Student).all()

    report = []
    for student in students:
        grades_report = check_grade_drop(db, student.id)
        attendance_report = check_unusual_absence(db, student.id)

        drops = [g for g in grades_report if g["status"] == "افت"]

        if drops or attendance_report["is_unusual"]:
            report.append({
                "student_id": student.id,
                "name": student.name,
                "class_name": student.class_name,
                "grade_drops": drops,
                "unusual_absence": attendance_report["is_unusual"],
                "total_absences": attendance_report["total_absences"],
            })

    return report

import os
from datetime import datetime, timedelta
from passlib.context import CryptContext
from jose import jwt, JWTError
from fastapi import Depends, HTTPException, Header

SECRET_KEY = os.getenv("SECRET_KEY", "school-ai-secret-change-me")
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 60 * 24

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def hash_password(password: str) -> str:
    return pwd_context.hash(password)


def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)


def create_access_token(data: dict) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)


def decode_access_token(token: str):
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        return payload
    except JWTError:
        return None


def get_current_user(authorization: str = Header(None)):
    if not authorization or not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="ورود الزامی است")

    token = authorization.split(" ")[1]
    payload = decode_access_token(token)

    if not payload:
        raise HTTPException(status_code=401, detail="نشست شما منقضی شده، دوباره وارد شوید")

    return payload


def require_parent(user: dict = Depends(get_current_user)):
    if user.get("role") != "parent":
        raise HTTPException(status_code=403, detail="دسترسی غیرمجاز")
    return user


def require_admin(user: dict = Depends(get_current_user)):
    if user.get("role") != "admin":
        raise HTTPException(status_code=403, detail="دسترسی غیرمجاز")
    return user

from sqlalchemy import Column, Integer, String, Float, Date, ForeignKey
from sqlalchemy.orm import relationship
from .database import Base


class Student(Base):
    __tablename__ = "students"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, nullable=False)
    class_name = Column(String, nullable=False)
    parent_name = Column(String, nullable=False)
    hashed_password = Column(String, nullable=True)   # ← این خط جدید

    grades = relationship("Grade", back_populates="student")
    attendances = relationship("Attendance", back_populates="student")
    


class Grade(Base):
    __tablename__ = "grades"

    id = Column(Integer, primary_key=True, index=True)
    student_id = Column(Integer, ForeignKey("students.id"), nullable=False)
    subject = Column(String, nullable=False)
    score = Column(Float, nullable=False)
    date = Column(Date, nullable=False)

    student = relationship("Student", back_populates="grades")


class Attendance(Base):
    __tablename__ = "attendances"

    id = Column(Integer, primary_key=True, index=True)
    student_id = Column(Integer, ForeignKey("students.id"), nullable=False)
    date = Column(Date, nullable=False)
    status = Column(String, nullable=False)  # "present" یا "absent"

    student = relationship("Student", back_populates="attendances")


class Exam(Base):
    __tablename__ = "exams"

    id = Column(Integer, primary_key=True, index=True)
    subject = Column(String, nullable=False)
    class_name = Column(String, nullable=False)
    date = Column(Date, nullable=False)


from sqlalchemy import Text, DateTime

@router.post("/scan")
def scan_school(db: Session = Depends(get_db), user: dict = Depends(require_admin)):
    issues = get_school_wide_report(db)
    created = []

    for item in issues:
        existing = (
            db.query(Report)
            .filter(Report.student_id == item["student_id"], Report.status == "pending")
            .first()
        )
        if existing:
            continue

        text = generate_report_text(
            db,
            student_name=item["name"],
            class_name=item["class_name"],
            drops=item["grade_drops"],
            total_absences=item["total_absences"],
        )

        report = Report(student_id=item["student_id"], content=text, status="pending")
        db.add(report)
        db.commit()
        db.refresh(report)
        created.append({"id": report.id, "student_name": item["name"], "content": text})

    return {"new_reports": created, "total_students_with_issues": len(issues)}

            "status": r.status,

import os
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from sqlalchemy.orm import Session

from ..auth import verify_password, create_access_token

router = APIRouter(prefix="/auth", tags=["Auth"])

ADMIN_USERNAME = os.getenv("ADMIN_USERNAME", "admin")
ADMIN_PASSWORD = os.getenv("ADMIN_PASSWORD", "admin1234")


class ParentLoginRequest(BaseModel):
    student_id: int
    password: str


class AdminLoginRequest(BaseModel):
    username: str
    password: str


@router.post("/parent/login")
def parent_login(data: ParentLoginRequest, db: Session = Depends(get_db)):
    student = db.query(Student).filter(Student.id == data.student_id).first()
    if not student or not student.hashed_password:
        raise HTTPException(status_code=401, detail="دانش‌آموز یا رمز عبور نادرست است")

    if not verify_password(data.password, student.hashed_password):
        raise HTTPException(status_code=401, detail="دانش‌آموز یا رمز عبور نادرست است")

    token = create_access_token({"role": "parent", "student_id": student.id})
    return {"access_token": token, "student_name": student.name}


@router.post("/admin/login")
def admin_login(data: AdminLoginRequest):
    if data.username != ADMIN_USERNAME or data.password != ADMIN_PASSWORD:

@router.post("/students/import")
async def import_students(
    file: UploadFile = File(...),
    db: Session = Depends(get_db),
    user: dict = Depends(require_admin),
):
    """
    فایل اکسل باید ۴ ستون داشته باشد به این ترتیب (بدون هدر یا با هدر، مهم نیست):
    نام | کلاس | نام والد | رمز عبور
    """
    content = await file.read()
    workbook = openpyxl.load_workbook(io.BytesIO(content))
    sheet = workbook.active

    created = []
    errors = []

    for i, row in enumerate(sheet.iter_rows(min_row=1, values_only=True), start=1):
        if not row or not row[0]:
            continue
        if str(row[0]).strip() in ("نام", "name", "Name"):
            continue

        try:
            name = str(row[0]).strip()
            class_name = str(row[1]).strip()
            parent_name = str(row[2]).strip()
            password = str(row[3]).strip() if len(row) > 3 and row[3] else "1234"

            student = Student(
                name=name,
                class_name=class_name,
                parent_name=parent_name,
                hashed_password=hash_password(password),
            )
            db.add(student)
            db.commit()
            db.refresh(student)
            created.append({"row": i, "name": name})
        except Exception as e:
            db.rollback()
            errors.append({"row": i, "error": str(e)})

    return {"created": created, "errors": errors, "total_created": len(created)}
